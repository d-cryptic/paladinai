```proto
syntax = "proto3";

import "google/protobuf/struct.proto";

service HubService {
  rpc DiscoverTools(DiscoverRequest)   returns (DiscoverResponse);

  repeated ToolSchema tools = 1;
}

message RegisterRequest {
  string integration_name = 1;
  string tenant_id        = 2;
  repeated ToolSchema tools = 3;
}

message RegisterResponse {
  repeated string tool_ids = 1;
}

message ListToolsRequest {
  string tenant_id = 1;
  string category  = 2;  // optional filter

}

message ToolMeta {
  string tool_id        = 1;
  string name           = 2;
  string integration    = 3;
  string category       = 4;
  string mcp_endpoint   = 5;
}

message ListToolsResponse {
  repeated ToolMeta tools = 1;
}

  string tool_id  = 1;
```

## 3. MCP Server Template

```go
// pkg/mcp/server.go — base for every integration MCP server

type MCPServer struct {
    name        string
    tools       map[string]ToolHandler
    vault       *vault.Client
    rateLimiter *rate.Limiter  // per-tenant, keyed by tenant_id
    hubConn     *grpc.ClientConn
}
// Scale policy declared in integration.yaml — see Section 10
// Default: scale_to_zero. Override to "always_on" for high-frequency integrations.

type ToolHandler func(ctx context.Context, tenantID string, params json.RawMessage) (json.RawMessage, error)

func NewMCPServer(name string, vault *vault.Client, hubAddr string) *MCPServer

// Startup sequence
func (s *MCPServer) Start(port int) error {
    s.registerTools()         // calls hub.RegisterTool for each tool
    s.startHealthEndpoint()   // GET /healthz → 200
    return s.serveGRPC(port)
}

// Every tool call goes through this
func (s *MCPServer) CallTool(ctx context.Context, req *mcp.CallToolRequest) (*mcp.CallToolResponse, error) {
    tenantID := extractTenant(ctx)          // from gRPC metadata
    creds, err := s.vault.GetSecret(        // dynamic credentials per tenant
        fmt.Sprintf("secret/tenants/%s/%s", tenantID, s.name),
    )
    if err != nil { return nil, ErrUnauthorized }

    if !s.rateLimiter.Allow() {             // per-tenant bucket
        return nil, ErrRateLimited
    }

    handler, ok := s.tools[req.ToolName]
    if !ok { return nil, ErrUnknownTool }

    result, err := handler(ctx, tenantID, req.Params)
    if err != nil { return nil, err }

    logToolCall(ctx, tenantID, s.name, req.ToolName, req.Params) // → ClickHouse
    return &mcp.CallToolResponse{Content: result}, nil
}

// Tool registration — description must include use_when + do_not_use_when
func (s *MCPServer) RegisterTool(t ToolSchema) {
    s.tools[t.Name] = t.Handler
    // pushed to hub on Start()
}
```

## 6. AlertEnvelope Normalization

```go
type AlertEnvelope struct {
    ID          string            // deduplication key
    Source      string            // "alertmanager" | "datadog" | "cloudwatch" | "pagerduty" | "webhook"
    TenantID    string
    Severity    string            // "critical" | "high" | "medium" | "low" | "info"
    Title       string
    Description string
    Labels      map[string]string
    StartsAt    time.Time
    EndsAt      *time.Time        // nil = still firing
    GeneratorURL string
    RawPayload  json.RawMessage   // original, unmodified
}
```

| Source          | Key Fields Mapped                                                | Normalization Notes                                    |
|-----------------|------------------------------------------------------------------|--------------------------------------------------------|
| Alertmanager    | labels.alertname→Title, annotations.summary→Description, startsAt, endsAt | severity from labels.severity              |
| Datadog         | title, message, priority→Severity, date_happened→StartsAt       | priority: 1=critical,2=high,3=medium,4=low             |
| CloudWatch SNS  | AlarmName→Title, NewStateReason→Description, StateChangeTime    | Severity inferred from AlarmName prefix or default=high|
| PagerDuty       | title, body.details→Description, severity, created_at           | direct mapping; status=resolved sets EndsAt            |
| Custom Webhook  | configurable field mapping in integration.yaml                   | fallback: description=raw JSON if unmatched             |

One Go file per source: `paladin-ingest/normalizer/alertmanager.go`, `datadog.go`, etc.
Router in `paladin-ingest/router.go` matches `Content-Type` + path prefix to normalizer.

## 9. Multi-tenant Tool Isolation

**Hub — DiscoverTools:**
```sql
-- filters to tenant's enabled integrations before vector search
SELECT t.tool_id, t.mcp_endpoint
FROM tools t
JOIN tenant_integration_permissions p
  ON t.integration_name = p.integration_name
WHERE p.tenant_id = $1 AND p.enabled = true
```
Qdrant search scoped to returned `tool_id` list via payload filter.

**MCP Server — every call:**
1. Extract `tenant_id` from gRPC metadata header `x-paladin-tenant-id`.
2. Fetch credentials: `vault.KVv2.Get("secret/tenants/{tenant_id}/{integration}")`.
3. Reject if secret missing or tenant_id mismatch.

Rate limit value comes from integration.yaml `rate_limit_per_tenant_per_minute` field, not hardcoded. Default: 100 req/min if not specified.

**Audit log — ClickHouse:**
```sql
CREATE TABLE tool_audit_log (
    ts          DateTime64(3),
    tenant_id   String,
    tool_name   String,
    params_hash String,   -- SHA-256 of params (no PII stored)
    duration_ms UInt32,
    success     Bool
) ENGINE = MergeTree()
ORDER BY (tenant_id, ts);
```

## 10. Scale Policy & Cold Start

### Tier Classification (declared in integration.yaml)

```yaml
deployment:
  scale_policy: "always_on"  # or "scale_to_zero"
  min_replicas: 1            # if always_on
  max_replicas: 5
  rate_limit_per_tenant_per_minute: 100  # default; override per integration
```

### Always-on (min_replicas: 1)

| Integration | Reason |
|---|---|
| mcp-prometheus | Used in every triage |
| mcp-loki | Used in every triage |
| mcp-alertmanager | Used in every alert ingestion |
| mcp-datadog | Primary metrics source if enabled |
| mcp-cloudwatch | Primary metrics source if enabled |
| mcp-slack | Every incident has Slack comms |
| mcp-pagerduty | Every P1/P2 has PD interaction |
| mcp-k8s | Every Kubernetes triage |

### Scale-to-zero (min_replicas: 0, KEDA-scaled)

| Integration | Reason |
|---|---|
| mcp-jira | Only when filing/updating tickets |
| mcp-linear | Same |
| mcp-confluence | Only for runbook ingestion (batch) |
| mcp-github | Only for code correlation (Stage 15) |
| mcp-newrelic | If not the primary metrics source |

### Speculative Pre-warm

To avoid cold-start latency:
1. At T+50ms, Supervisor classifier outputs intent + likely tools needed
2. If a scale-to-zero integration is predicted: fire `GET /healthz` on its MCP service
3. KEDA wakes the pod (~3-5s)
4. By the time agent actually needs the tool (T+5-10s), pod is ready
5. Cold-start hidden in the parallel pre-warm window

## 11. Webhook Signature Verification

Different sources sign webhooks differently. One verifier per source.

| Source | Header | Algorithm | Notes |
|---|---|---|---|
| Alertmanager | `X-Paladin-Signature` | HMAC-SHA256 (Paladin-issued secret) | Alertmanager has no native signing — Paladin issues secret on integration enable |
| Datadog (webhook) | `X-Paladin-Signature` | HMAC-SHA256 (Paladin-issued secret) | Same — Datadog webhooks not natively signed |
| PagerDuty | `X-PagerDuty-Signature` | HMAC-SHA256 (PD secret) | Verify against PD-provided secret |
| GitHub | `X-Hub-Signature-256` | HMAC-SHA256 timing-safe | Timing-safe compare mandatory |
| Slack | `X-Slack-Signature` | HMAC-SHA256 + timestamp | Timestamp check + signature |
| Custom Webhook | `X-Paladin-Signature` | HMAC-SHA256 (Paladin-issued) | Tenant configures their tool to send this header |

### Implementation

One file per source: `paladin-ingest/normalizer/{source}/verify.go`. Each implements:

```go
type Verifier interface {
    Verify(ctx context.Context, headers http.Header, body []byte) error
}
```

Verifier extracts the signing secret from Vault keyed by tenant_id. On verification failure: 401, log to ClickHouse audit, drop request.

### Replay Protection

HMAC alone doesn't prevent replay attacks. Add for ALL sources:

1. **Timestamp check**: reject if `now - webhook_timestamp > 5min`
2. **Nonce dedup**: store `(tenant_id, webhook_id, timestamp)` in Valkey, TTL 10min
3. **First-seen-only**: if nonce already seen, reject with 409

```go
func (v *Verifier) checkReplay(ctx context.Context, tenantID, webhookID string, ts time.Time) error {
    if time.Since(ts) > 5*time.Minute {
        return ErrTimestampTooOld
    }
    key := fmt.Sprintf("nonce:%s:%s:%d", tenantID, webhookID, ts.Unix())
    ok, err := v.valkey.SetNX(ctx, key, 1, 10*time.Minute).Result()
    if err \!= nil {
        return err
    }
    if \!ok {
        return ErrReplayDetected
    }
    return nil
}
```

```

        │
        ▼

        │
        ▼

        │
        ▼

  1. Verify state token against Valkey (CSRF protection)
  2. Exchange code for access_token + refresh_token (GitHub API)
  3. Store in Vault: secret/tenants/{tid}/github

        │
        ▼

✓ github enabled — 4 tools registered
```

### Token Refresh

Background job runs every 15min:
```go
func (r *OAuthRefresher) Run(ctx context.Context) {
    expiring := r.vault.ListExpiringTokens(ctx, 30*time.Minute)
    for _, token := range expiring {
        newToken, err := r.refreshAccessToken(token.RefreshToken, token.Provider)
        if err \!= nil {
            r.alertSlack(token.TenantID, "OAuth refresh failed: "+err.Error())
            continue
        }
        r.vault.UpdateSecret(ctx, token.Path, newToken)
    }
}
```

## 13. Outbound Webhook Reliability

MCP servers like mcp-slack, mcp-pagerduty, mcp-linear SEND data outbound. Reliability matters.

### Retry Policy (per outbound call)

```go
const (
    OutboundMaxRetries = 3
    OutboundBackoff    = []time.Duration{1*time.Second, 5*time.Second, 30*time.Second}
)

func (s *MCPServer) sendWithRetry(ctx context.Context, req *http.Request) error {
    for attempt := 0; attempt <= OutboundMaxRetries; attempt++ {
        resp, err := s.httpClient.Do(req)
        if err == nil && resp.StatusCode < 500 {
            return nil
        }
        if attempt < OutboundMaxRetries {
            time.Sleep(OutboundBackoff[attempt])
        }
    }
    return s.publishToDLQ(ctx, req)
}
```

### DLQ for Failed Outbound

Failed sends → NATS subject `dlq.outbound.{integration}.{tenant_id}`.
Retained 30d.

### Operator Alert

Per-tenant DLQ depth metric in Prometheus:
```
paladin_outbound_dlq_depth{tenant_id, integration}
```

Alert: depth > 10 for 5min → page on-call. Common causes:
- Slack workspace deactivated
- PagerDuty service disabled
- OAuth token expired (caught earlier by refresh job, but defense in depth)

### Idempotency Keys

For mcp-pagerduty.create_incident, mcp-linear.create_issue: include idempotency key in request:
```go
req.Header.Set("Idempotency-Key", fmt.Sprintf("%s-%s", tenantID, alertFingerprint))
```
Prevents duplicate creation if retries happen after partial success.

## 14. Tool Versioning & Upgrade Strategy

### Hub Stores Versioned Tools

When mcp-datadog 1.0.0 → 1.1.0, tools may be renamed or schema changed.

PG schema:
```sql
CREATE TABLE hub_tools (
    tool_id          UUID PRIMARY KEY,
    integration      TEXT NOT NULL,
    integration_ver  TEXT NOT NULL,
    name             TEXT NOT NULL,
    version          TEXT NOT NULL,
    description      TEXT NOT NULL,
    use_when         TEXT,
    do_not_use_when  TEXT,
    input_schema     JSONB,
    output_schema    JSONB,
    deprecated_at    TIMESTAMPTZ,           -- set when superseded
    superseded_by    UUID REFERENCES hub_tools(tool_id),
    registered_at    TIMESTAMPTZ DEFAULT now()
);
```

### Tool Identity in Runbooks

Runbooks reference tools by versioned name:
```
prometheus.query_range@v1     ← exact version
prometheus.query_range        ← latest active version
```

Hub resolves unqualified names to latest active.

### Upgrade Flow

```
mcp-datadog 1.1.0 deploys (replaces 1.0.0):
  1. New pod registers tools with new version
  2. Hub: INSERT new tool rows with version=1.1.0
  3. Hub: SET deprecated_at=now() on tools from 1.0.0
  4. Hub: SET superseded_by linking old → new
  5. Both versions queryable for 30 days (grace period)
  6. After 30d: hard-deprecate (DiscoverTools excludes them entirely)
  7. Runbooks pinning @v1 explicit version still work until grace expires
```

### Breaking Schema Changes

If input_schema is incompatible: bump major version (1.0 → 2.0). Old version kept available for 90d, alerts sent to tenants using deprecated tools.

## 16. Auto-Detection for `paladin init`

When CLI runs `paladin init`, auto-detect what's running in the cluster.

### Detection Probes (per integration)

```yaml
# integration.yaml
detection:
  probes:
    - type: kubernetes_service
      pattern: "prometheus.*"
      namespaces: ["monitoring", "kube-system", "observability"]
    - type: helm_release
      chart: "prometheus*"
    - type: http_get
      url_template: "http://{service}.{ns}.svc.cluster.local:9090/api/v1/labels"
      success_codes: [200]
```

### Detection Flow

```go
func (d *Detector) DetectIntegrations(ctx context.Context) []DetectedIntegration {
    var detected []DetectedIntegration
    for _, integration := range d.knownIntegrations {
        for _, probe := range integration.DetectionProbes {
            if probe.Match(ctx, d.k8sClient) {
                detected = append(detected, DetectedIntegration{
                    Name:       integration.Name,
                    Confidence: probe.Confidence,
                    Hint:       probe.Hint,  // e.g., "found at prometheus.monitoring.svc"
                })
                break
            }
        }
    }
    return detected
}
```

### Specific Probes per Integration

| Integration | Probe |
|---|---|
| Prometheus | `kubectl get svc -A -l app.kubernetes.io/name=prometheus` |
| Loki | `kubectl get svc -A -l app.kubernetes.io/name=loki` |
| Datadog | `kubectl get ds -A -l app.kubernetes.io/name=datadog-agent` |

Until tenant-uploaded MCP integrations ship in v2.1, customers can integrate any internal tool via the generic webhook integration.

### Inbound (alerts FROM customer's internal tool)

```yaml
$ paladin integrations enable custom-webhook
> Generated webhook URL: https://api.paladinai.io/webhook/acme/inbound/abc123
> HMAC secret: <displayed once>
> Configure your tool to POST alerts to the URL above with X-Paladin-Signature header
```

Inbound webhook payload follows AlertEnvelope schema (or includes a transformation script for arbitrary JSON → AlertEnvelope mapping).

### Outbound (Paladin sends events TO customer's internal tool)

```yaml
$ paladin integrations enable custom-webhook --outbound-url=https://internal.acme.com/paladin
> HMAC secret stored in Vault
> Paladin will POST events to your URL with X-Paladin-Signature header
```

Outbound events:
- `incident.created`
- `incident.classified`
- `incident.rca_ready`
- `incident.resolved`

Schema documented in stage 20 (External API).

### What This Doesn't Cover

- Tenant cannot run custom CODE inside Paladin's environment
- Tenant cannot expose CUSTOM TOOLS to the agent
- For those: wait for v2.1 with sandboxed MCP execution

## 18. Health Propagation (Two Layers)

### Layer 1 — Per-Tool Circuit Breaker (gobreaker)

Inside each MCP server. Per-tool state machine:
```
Closed → Open: 50% errors over 60s rolling window
Open → Half-Open: 30s cooldown
Half-Open → Closed: 3 successful probes
Half-Open → Open: any failure
```

Tool-level state pushed to hub on every state change:
```go
hub.UpdateToolHealth(ctx, &ToolHealth{
    ToolID:  toolID,
    State:   "open",
    Reason:  "external_api_5xx",
    OpenedAt: time.Now(),
})
```

### Layer 2 — Periodic External Dependency Probe

MCP servers run a 30s ticker that probes their external dependency:

```go
func (s *MCPServer) dependencyProbe(ctx context.Context) {
    ticker := time.NewTicker(30 * time.Second)
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            healthy := s.probe()  // integration-specific
            s.hub.UpdateDependencyHealth(ctx, &DepHealth{
                Integration: s.name,
                Healthy:     healthy,
                CheckedAt:   time.Now(),
            })
        }
    }
}
```

Lightweight probes per integration:
| Integration | Probe |
|---|---|
| mcp-datadog | `GET /api/v1/validate` (validates API key) |
| mcp-pagerduty | `GET /users?limit=1` |
| mcp-slack | `auth.test` |
| mcp-prometheus | `GET /api/v1/labels?limit=1` |
| mcp-loki | `GET /loki/api/v1/labels?limit=1` |
| mcp-github | `GET /user` |

### Hub Combines Both Signals

```go
func (h *Hub) ToolHealthy(toolID string) bool {
    tool := h.lookupTool(toolID)
    return tool.PodAlive &&
           tool.DependencyHealthy &&
           tool.CircuitState \!= "open"
}
```

`DiscoverTools` excludes tools where `ToolHealthy(tool_id) == false`.

## 19. Tool Description Quality Enforcement

Tool descriptions directly drive hub semantic search accuracy. Lazy descriptions cause tool selection failure.

### CI-Time Validation

When an integration package is built, validate every tool description:

```go
type DescriptionLinter struct{}

func (l *DescriptionLinter) Lint(tool ToolSchema) []LintIssue {
    var issues []LintIssue
    
    // Length checks
    if len(tool.Description) > 100 {
        issues = append(issues, LintIssue{Severity: "error", Msg: "description > 100 chars"})
    }
    if len(tool.UseWhen) < 20 || len(tool.UseWhen) > 200 {
        issues = append(issues, LintIssue{Severity: "error", Msg: "use_when length 20-200 chars"})
    }
    
    // Lazy phrase detection
    lazyPhrases := []string{"when you need data", "for general purposes", "anytime", "always"}
    for _, phrase := range lazyPhrases {
        if strings.Contains(strings.ToLower(tool.UseWhen), phrase) {
            issues = append(issues, LintIssue{
                Severity: "error",
                Msg:      "use_when contains lazy phrase: " + phrase,
            })
        }
    }
    
    // Mandatory fields
    if tool.UseWhen == "" || tool.DoNotUseWhen == "" {
        issues = append(issues, LintIssue{Severity: "error", Msg: "use_when and do_not_use_when mandatory"})
    }
    
    // Verb-first check on description
    firstWord := strings.SplitN(tool.Description, " ", 2)[0]
    if \!isVerb(firstWord) {
        issues = append(issues, LintIssue{Severity: "warn", Msg: "description should start with verb"})
    }
    
    return issues
}
```

### CI Pipeline Step

Every integration package PR runs:
```bash
paladin lint integration.yaml
```

Build fails on any error-severity lint issue.

### Discrimination Test

For each pair of tools, run a similarity check on use_when:
- If two tools' use_when are too similar (cosine > 0.85 in embedding space): warning
- Forces authors to write distinguishable descriptions

This is checked at registration time (hub rejects ambiguous tools).
